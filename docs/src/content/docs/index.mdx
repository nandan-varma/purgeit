import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Safe by design" icon="approve-check">
    Nothing is selected by default, nothing is deleted without confirmation, and every path is re-verified against the rules, git, and recent activity right before it is removed.
  </Card>
  <Card title="Fast" icon="rocket">
    One traversal gives size, file count and last activity. Hardlinked files (pnpm) are counted only when deleting them really frees space.
  </Card>
  <Card title="Interactive UI" icon="screen">
    A full-screen terminal UI that fills in as measurements arrive: sort, filter, select, confirm.
  </Card>
  <Card title="Scriptable" icon="document">
    `scan --format json` produces a report you can review and feed back to `clean --plan`. Stable exit codes.
  </Card>
  <Card title="65 rules, 16 ecosystems" icon="setting">
    JS/TS, Python, Rust, Go, Apple, JVM, .NET and more — extend them with a typed, JSON-Schema-validated config.
  </Card>
  <Card title="A library too" icon="puzzle">
    `scan`, `clean`, and the rule engine are plain async-iterator APIs with no UI dependency.
  </Card>
</CardGrid>

```bash
npx purgeit ~/dev          # browse and select
npx purgeit scan ~/dev     # just list
npx purgeit clean ~/dev    # delete, with confirmation
```
